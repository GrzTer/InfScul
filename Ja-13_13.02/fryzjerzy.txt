SELECT Klienci.id_klienta
FROM Uslugi INNER JOIN (Klienci INNER JOIN ((Salony INNER JOIN Fryzjerzy ON Salony.id_salonu = Fryzjerzy.id_salonu) INNER JOIN Wizyty ON Fryzjerzy.id_fryzjera = Wizyty.id_fryzjera) ON Klienci.id_klienta = Wizyty.id_klienta) ON Uslugi.id_uslugi = Wizyty.id_uslugi
GROUP BY Klienci.id_klienta, Salony.nazwa_salonu
HAVING (((Salony.nazwa_salonu)="Magnolia"));
